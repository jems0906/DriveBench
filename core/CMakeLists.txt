add_library(drivebench src/kinematics.cpp src/collision.cpp src/world.cpp src/scenario.cpp
  src/driver.cpp src/evaluator.cpp src/recorder.cpp src/runner.cpp)
target_link_libraries(drivebench PRIVATE nlohmann_json::nlohmann_json)
target_include_directories(drivebench PUBLIC include)
target_compile_features(drivebench PUBLIC cxx_std_20)
set_target_properties(drivebench PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(drivebench PRIVATE /W4 /fp:strict)
else()
  target_compile_options(drivebench PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()

if(BUILD_TESTING)
  add_executable(drivebench_tests tests/test_core.cpp)
  target_link_libraries(drivebench_tests PRIVATE drivebench)
  add_test(NAME kinematics COMMAND drivebench_tests)
  add_executable(drivebench_acceptance tests/test_acceptance.cpp)
  target_link_libraries(drivebench_acceptance PRIVATE drivebench)
  foreach(check IN ITEMS metrics determinism seed_sweep verdicts harsh_suite regression validation feedback)
    add_test(NAME ${check} COMMAND drivebench_acceptance ${check} "${PROJECT_SOURCE_DIR}/scenarios")
  endforeach()
endif()
